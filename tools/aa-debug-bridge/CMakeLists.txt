cmake_minimum_required(VERSION 3.16)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Compatibility for aasdk's gitversion module")
project(carplayd_aa_debug_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_AA_BRIDGE "Build the USB Android Auto bridge" ON)
set(AASDK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../aasdk" CACHE PATH "opencardev/aasdk checkout")
find_package(Boost REQUIRED COMPONENTS system log log_setup)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
# Force the same compatibility definitions into aasdk and its consumers.
add_library(bridge_core core.cpp debug_client.cpp)
target_compile_options(bridge_core PUBLIC -include "${CMAKE_CURRENT_SOURCE_DIR}/boost_compat.hpp")
target_include_directories(bridge_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(bridge_core PUBLIC Boost::system nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_AA_BRIDGE)
    if(NOT EXISTS "${AASDK_SOURCE_DIR}/include/aasdk/Messenger/Messenger.hpp")
        message(FATAL_ERROR "Set AASDK_SOURCE_DIR to the opencardev/aasdk checkout")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(BRIDGE_USB REQUIRED IMPORTED_TARGET libusb-1.0)
    find_program(BRIDGE_PROTOC protoc REQUIRED)
    set(Protobuf_PROTOC_EXECUTABLE "${BRIDGE_PROTOC}" CACHE FILEPATH "System protoc" FORCE)
    find_package(Protobuf REQUIRED)
    set(AASDK_TEST OFF CACHE BOOL "" FORCE)
    set(AASDK_BENCHMARK OFF CACHE BOOL "" FORCE)
    set(SKIP_BUILD_PROTOBUF ON CACHE BOOL "" FORCE)
    set(SKIP_BUILD_ABSL ON CACHE BOOL "" FORCE)
    add_subdirectory("${AASDK_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/aasdk" EXCLUDE_FROM_ALL)
    # This executable only uses USB. Upstream's unused TCP implementation still
    # calls address::from_string(), removed from current Boost.
    get_target_property(BRIDGE_AASDK_SOURCES aasdk SOURCES)
    list(FILTER BRIDGE_AASDK_SOURCES EXCLUDE REGEX "/TCP/|/Transport/TCPTransport[.]cpp$")
    set_property(TARGET aasdk PROPERTY SOURCES "${BRIDGE_AASDK_SOURCES}")
    target_compile_options(aasdk PUBLIC -include "${CMAKE_CURRENT_SOURCE_DIR}/boost_compat.hpp")
    # aasdk's source include path is directory-scoped upstream.
    target_include_directories(aasdk PUBLIC "$<BUILD_INTERFACE:${AASDK_SOURCE_DIR}/include>" "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/aasdk/include>")
    add_executable(aa-debug-bridge main.cpp session.cpp)
    target_link_libraries(aa-debug-bridge PRIVATE bridge_core aasdk PkgConfig::BRIDGE_USB)
    target_compile_options(aa-debug-bridge PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(aa-protocol-tests protocol_tests.cpp)
    target_link_libraries(aa-protocol-tests PRIVATE bridge_core aasdk)
    add_test(NAME aa-protocol-tests COMMAND aa-protocol-tests)
endif()
enable_testing()
add_executable(bridge-tests tests.cpp)
target_link_libraries(bridge-tests PRIVATE bridge_core)
add_test(NAME bridge-tests COMMAND bridge-tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME debug-network COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_network.py" "$<TARGET_FILE:bridge-tests>")
    # The desktop receiver standing in for carplayd, driven by the same C++ client.
    add_test(NAME debug-receiver COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../test_debug_receiver.py")
    set_tests_properties(debug-receiver PROPERTIES ENVIRONMENT "BRIDGE_TESTS=$<TARGET_FILE:bridge-tests>")
endif()
